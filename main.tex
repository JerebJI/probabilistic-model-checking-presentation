\begin{frame}{Motivation}
    We want to model a probabilistic system and check its properties.

    Examples include:
    \begin{itemize}
        \item Randomised distributed algorithms
        \item Communication, network and multimedia protocols
        \item Security protocols
        \item Biological systems
        \item Planning and synthesis
        \item Performance and reliability
        \item Game theory
        \item Power management
        \item ...
    \end{itemize}
\end{frame}

\begin{frame}{How do you model a probabilistic system?}
    \begin{enumerate}
        \item describe states and transitions
        \item make number of state finite by abstraction
        \item assign probabilities to transitions
    \end{enumerate}
\end{frame}

\begin{frame}{Formalize the model}
    \begin{itemize}
        \item \textbf{Set of states} $$S = \{\text{Needs\_met}, \text{Needs\_not\_met}, \text{Competitor}, \:\dots\}$$
        \item \textbf{Initial state} $$s_\text{start} = \text{Needs\_met}$$
        \item \textbf{Transition probability matrix} \\[0.5em]
              \hspace{2em} $P : S \times S \rightarrow [0,1]$ \\
              \hspace{2em} where $\sum_{s' \in S} P(s,s') = 1$ for all $s \in S$

    \end{itemize}
\end{frame}

\begin{frame}{Formalize the model}

    \[
        \begin{array}{c@{\;}c}
            % Row labels
            \begin{array}{c}
                \\[-0.3em]
                \text{Needs\_met}      \\
                \text{Needs\_not\_met} \\
                \vdots
            \end{array}
             &
            % Column labels + matrix
            \begin{array}{c}
                \text{Needs\_met} \hspace{3em} \text{Needs\_not\_met} \cdots
                \\[0.3em]
                \left(
                \begin{array}{cccc}
                        \hspace{2em} 0.5 \hspace{4em}      & 0.5 \hspace{4em} & \cdots \\
                        \hspace{2em} 0 \hspace{4em}        & 0 \hspace{4em}   & \cdots \\
                        \hspace{2em} 1.0      \hspace{4em} & 0 \hspace{4em}   & \cdots \\
                        \vdots   \hspace{2em}             & \vdots   \hspace{4em}         & \ddots
                    \end{array}
                \right)
            \end{array}
        \end{array}
    \]
\end{frame}

\begin{frame}{Continuous-time Markov chains}

        \item \textbf{Transition rate matrix} \\[0.5em]
              \hspace{2em} $Q : S \times S \rightarrow [0,\infty)$

    \end{itemize}

\begin{frame}{Formalize nondeterminism}
    % formal definition of a MDP
    \begin{itemize}
        \item \textbf{Set of states} $$S = \{\text{Needs\_met}, \text{Advertisement}, \text{Competitor}, \:\dots\}$$
        \item \textbf{Initial state} $$s_\text{start} = \text{Needs\_met}$$
        \item \textbf{Transition probability function} \\[0.5em]
               $T : S \rightarrow 2^{\text{Dist}(S)}$ \\
                \hspace{2em} $\begin{aligned}T(\text{Product\_view}):=\{&\textcolor{CadetBlue}{\{Advertisement\mapsto 0.7,Competitor\mapsto 0.3\}},\\
                &\textcolor{YellowOrange}{\{Checkout\mapsto 0.7,Competitor\mapsto 0.3\}}\}\end{aligned}$
    \end{itemize}
\end{frame}

\begin{frame}{CTL vs PCTL}
    \textcolor{lightgray}{CTL - Computation Tree Logic }\\
    PCTL - \textbf{Probabilistic} Computation Tree Logic\\
    \begin{tabular}{lp{30mm}p{30mm}}
        addition: probabilistic operator $P$
    \end{tabular}
\end{frame}
\begin{frame}{CTL vs PCTL - definitions}
    \begin{itemize}
        \item State formulae:
              \begin{itemize}
                  \item[] \textcolor{lightgray}{$\phi ::= \text{true} \mid a \mid \phi \land \phi \mid \neg \phi \mid \forall\,\psi \mid \exists,\psi$}
                  \item[]
                      $\phi ::= \text{true} \mid a \mid \phi \land \phi \mid \neg \phi \mid P_{\sim p}[\psi]$,
                  \item[] where $a$ is an atomic proposition,
                      $p \in [0,1]$ is a probability bound,
                      $\sim \in \{<,>,\leq,\geq\}$,
                      $k \in \mathbb{N}$
              \end{itemize}

        \item Path formulae:
              \begin{itemize}
                  \item[] \textcolor{lightgray}{$\psi ::= X\,\phi \mid F\,\phi \mid G\,\phi \mid \phi \, U \, \phi$}
                  \item[] \textcolor{lightgray}{where $X$ means "next", $G$ means "globally", and $U$ means "until"}
                  \item[]
                      $\psi ::= X\,\phi
                          \;\mid\;
                          \phi\, U^{\leq k}\, \phi
                          \;\mid\;
                          \phi\, U\, \phi$,\\

                  \item[] where $X$ means "next", $U$ means "until", and $U^{\leq k}$ means "bounded until"\\
                      \hfill {(path formulae only occur inside the P operator)}

              \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{CTL vs PCTL - semantics of state formulae}
    \begin{itemize}

        \item[] $s \models \phi$ denotes ``$s$ satisfies $\phi$'' or ``$\phi$ is true in $s$''

            \vspace{1em}

        \item[] For a state $s$ of an LTS $(S,s_{\text{init}},\rightarrow,L)$:
            \begin{align*}
                s \models \text{true}
                 & \qquad \text{always}                                                                          \\[0.5em]
                s \models a
                 & \iff a \in L(s)                                                                               \\[0.5em]
                s \models \phi_1 \land \phi_2
                 & \iff s \models \phi_1 \text{ and } s \models \phi_2                                           \\[0.5em]
                s \models \neg \phi
                 & \iff s \not\models \phi                                                                       \\[0.5em]
                \textcolor{lightgray}{s \models \forall \psi}
                 & \textcolor{lightgray}{\iff \omega \models \psi \text{ for all } \omega \in \mathrm{Path}(s)}  \\[0.5em]
                \textcolor{lightgray}{s \models \exists \psi}
                 & \textcolor{lightgray}{\iff \omega \models \psi \text{ for some } \omega \in \mathrm{Path}(s)}
            \end{align*}
    \end{itemize}
    \text{(we will mention operators $\forall$ and $\exists$ in PCTL later)}
\end{frame}
\begin{frame}{CTL vs PCTL - semantics of path formulae}

    \begin{itemize}
        \item[] $\omega \models \psi$ denotes ``$\omega$ satisfies $\psi$'' or ``$\psi$ is true along $\omega$''
            \vspace{1em}

        \item[] For a path $\omega$ of an LTS $(S,s_{\text{init}},\rightarrow,L)$:
            \begin{align*}
                \omega \models X\,\phi
                 & \iff \omega(1) \models \phi                                                        \\[0.5em]
                \textcolor{lightgray}{\omega \models F\,\phi}
                 & \textcolor{lightgray}{\iff \exists k \geq 0 \text{ s.t. } \omega(k) \models \phi } \\[0.5em]
                \textcolor{lightgray}{\omega \models G\,\phi}
                 & \textcolor{lightgray}{\iff \forall i \geq 0 \; \omega(i) \models \phi }            \\[0.5em]
                \omega \models \phi_1 \, U \, \phi_2
                 & \iff \exists k \geq 0 \text{ s.t. } \omega(k) \models \phi_2                       \\
                 & \qquad \text{and } \forall i < k \; \omega(i) \models \phi_1                       \\
                \omega \models \phi_1 \, U^{\leq k} \, \phi_2
                 & \;\iff\;
                \exists i \leq k \text{ such that } \omega(i) \models \phi_2                          \\
                 & \qquad \text{and } j < i,\; \omega(j) \models \phi_1
            \end{align*}

    \end{itemize}

\end{frame}

\begin{frame}{PCTL - semantics of the probability operator}

    \begin{itemize}

        \item[] Notation:
            \[
                s \models P_{\sim p}[\psi]
            \]
            means that the probability, from state $s$, that $\psi$ is true for an outgoing path satisfies $\sim p$.

            \vspace{1em}

        \item[] Example:
            \[
                s \models P_{<0.15}[X\ \text{succ}]
            \]
            $\iff$
            ``the probability of succeed in the next state of outgoing paths from $s$ is less than $0.15$''

        \item[] formally:
            \[
                s \models P_{\sim p}[\psi]
                \iff
                {\mathrm{Prob}(s,\psi)\sim p}
            \]
        \item[] where:
            \[
                {\mathrm{Prob}(s,\psi)}
                =
                \Pr_s \left\{
                \omega \in \mathrm{Path}(s)
                \mid
                \omega \models \psi
                \right\}
            \]

    \end{itemize}

\end{frame}

\begin{frame}{PCTL - basic logical equivalences}

    \begin{itemize}
        \item[] False:
            \[
                \text{false} \equiv \neg \text{true}
            \]

        \item[] Disjunction:
            \[
                \phi_1 \lor \phi_2
                \equiv
                \neg(\neg \phi_1 \land \neg \phi_2)
            \]

        \item[] Implication:
            \[
                \phi_1 \rightarrow \phi_2
                \equiv
                \neg \phi_1 \lor \phi_2
            \]
        \item[] Negation example:
            \[
                \neg P_{>p}[\phi_1 \ U \ \phi_2]
                \equiv
                P_{\leq p}[\phi_1 \ U \ \phi_2]
            \]

    \end{itemize}

\end{frame}

\begin{frame}{PCTL - examples}

    \begin{itemize}
        \item "The probability that the sender has received $n$ acknowledgements within $k$ clock--ticks is at least $0.8$":
              \[
                  P_{\geq 0.8}\left[
                      F^{\leq k}\ \texttt{reply\_count}=n
                      \right]
              \]

        \item "The probability that component B fails before component A is less than $0.4$"
              \[
                  P_{<0.4}\left[
                      \neg \texttt{fail}_A \ U \ \texttt{fail}_B
                      \right]
              \]
    \end{itemize}

\end{frame}

\begin{frame}{PCTL - operators $\forall$ and $\exists$}
    We use $P$ operator to define $\forall$ and $\exists$.
    \begin{itemize}
        \item
              \textbf{Qualitative}
              PCTL properties

              \begin{itemize}
                  \item[]
                      $
                          P_{\sim p}[\psi],
                      $
                      where $p$ is either $0$ or $1$.
              \end{itemize}

        \item
              \textbf{Quantitative}
              PCTL properties

              \begin{itemize}
                  \item[]
                      $
                          P_{\sim p}[\psi],
                      $
                      where $p$ is in the range $(0,1)$.
              \end{itemize}

              \vspace{1em}
              $\implies$
              \begin{itemize}
                  \item[]
                      {
                      $P_{>0}[F\ \phi]$
                      }
                      is identical to
                      $\exists F\ \phi$

                      \begin{itemize}
                          \item[]
                              there exists a finite path to a $\phi$--state
                      \end{itemize}

                  \item[]
                      {
                      $P_{\geq 1}[F\ \phi]$
                      }
                      is (similar to but) weaker than
                      $\forall F\ \phi$

                      \begin{itemize}
                          \item[]
                              a $\phi$--state is reached ``almost surely''
                      \end{itemize}
              \end{itemize}
    \end{itemize}

\end{frame}

\begin{frame}{PCTL - limitations}

    \begin{itemize}

        \item[]
            PCTL is very useful for specifying
            \textbf{probabilistic reachability} properties

            \begin{itemize}
                \item[$-$]
                    e.g.\ ``what is the probability of eventually reaching
                    a failure/success state?''
                \item[$-$]
                    most PCTL properties are variations of:
                    \[
                        \phi \ U \ \psi
                    \]
                    meaning:
                    ``stay in $\phi$ states until reaching a $\psi$ state''
            \end{itemize}

        \item[]
            Therefore, PCTL mainly reasons about:

            \begin{itemize}
                \item[$-$]
                    reaching target states
                \item[$-$]
                    staying within allowed states beforehand
                \item[$-$]
                    probabilities of these events
                \item[$-$]
                    possibly within bounded time
            \end{itemize}

    \end{itemize}

\end{frame}

\begin{frame}{PCTL - limitations}

    \begin{itemize}
        \item[]
            However, PCTL has limited expressivity

            \begin{itemize}
                \item[$-$]
                    difficult to express complex long--term behaviours
                \item[$-$]
                    e.g.\ ``every request is eventually acknowledged''
                \item[$-$]
                    e.g.\ ``event $A$ happens infinitely often''
            \end{itemize}

        \item[]
            More expressive temporal logics include:

            \begin{itemize}
                % \item[$-$]
                %     LTL (Linear Temporal Logic)
                %     for richer path behaviours

                \item[$-$]
                    PCTL$^{*}$,
                    which combines probabilities with richer temporal reasoning
            \end{itemize}

    \end{itemize}

\end{frame}

\begin{frame}{How does PCTL model checking work?}

    Given a DTMC $\mathcal{D} = (S, s_{\text{init}}, \mathbf{P}, L)$
    and a PCTL formula $\phi$:

    \vspace{1em}

    \textbf{Goal:} compute the \emph{satisfaction set}
    $\mathrm{Sat}(\phi) = \{s \in S \mid s \models \phi\}$

    \vspace{1em}

    \textbf{Algorithm:} bottom-up on the syntax tree of $\phi$
    \begin{enumerate}
        \item Parse $\phi$ into subformulae
        \item Process from leaves to root:
              \begin{itemize}
                  \item Atomic props, $\lnot$, $\land$ \;:\;
                        same as CTL (set operations)
                  \item $P_{\sim p}[X\,\phi_1]$ \;:\;
                        \textbf{matrix--vector multiplication}
                  \item $P_{\sim p}[\phi_1\, U^{\leq k}\, \phi_2]$ \;:\;
                        \textbf{iterative} matrix--vector multiplication
                  \item $P_{\sim p}[\phi_1\, U\, \phi_2]$ \;:\;
                        \textbf{linear equation system}
              \end{itemize}
        \item Check whether $s_{\text{init}} \in \mathrm{Sat}(\phi)$
    \end{enumerate}

    \vspace{0.5em}
    \hfill{\scriptsize\textcolor{gray}{[Baier \& Katoen, Ch.~10; Parker, Lecture~5]}}
\end{frame}

\begin{frame}{PCTL model checking --- Next operator}

    \textbf{Goal:} check $P_{\sim p}[X\,\phi_1]$ for each state $s$.

    \vspace{0.5em}
    For each state $s$, compute the probability of reaching
    a $\phi_1$--state in one step:
    \[
        \mathrm{Prob}(s, X\,\phi_1)
        \;=\;
        \sum_{s' \in \mathrm{Sat}(\phi_1)} \mathbf{P}(s, s')
    \]

\end{frame}

\begin{frame}{}
    This is a single \textbf{matrix--vector product}:
    $\;\vec{x} = \mathbf{P} \cdot \vec{v}$,
    where $\vec{v}$ is the characteristic vector of $\mathrm{Sat}(\phi_1)$.

    \vspace{0.5em}

    \begin{center}
        \resizebox{0.85\textwidth}{!}{%

    iff $x_s \sim p$.
\end{frame}

\begin{frame}{PCTL model checking --- Bounded until}

    \textbf{Goal:} check $P_{\sim p}[\phi_1 \, U^{\leq k}\, \phi_2]$.

    \vspace{0.5em}

    Compute probability by \textbf{dynamic programming} (backward from step $k$):

    \[
        x_s^{(t)} =
        \begin{cases}
            1 & \text{if } s \models \phi_2                            \\
            0 & \text{if } s \not\models \phi_1 \text{ or } t = 0     \\
            \displaystyle\sum_{s' \in S} \mathbf{P}(s,s')\cdot x_{s'}^{(t-1)}
              & \text{otherwise}
        \end{cases}
    \]

\end{frame}

\begin{frame}{}
        \begin{center}
        \resizebox{0.9\textwidth}{!}{%
        \begin{tikzpicture}[
                box/.style={draw, rounded corners, minimum width=20mm,
                        minimum height=8mm, font=\small, align=center},
                arrow/.style={-{Stealth}, thick},
            ]
            \node[box] (k) at (0,0) {step $k$};
            \node[box] (k1) at (3.5,0) {step $k{-}1$};
            \node[box] (dots) at (7,0) {$\cdots$};
            \node[box] (z) at (10.5,0) {step $0$};

            \draw[arrow] (k) -- node[above, font=\footnotesize] {$\mathbf{P}\cdot\vec{x}$} (k1);
            \draw[arrow] (k1) -- node[above, font=\footnotesize] {$\mathbf{P}\cdot\vec{x}$} (dots);
            \draw[arrow] (dots) -- node[above, font=\footnotesize] {$\mathbf{P}\cdot\vec{x}$} (z);

        \end{tikzpicture}
        }
    \end{center}

    $k$ iterations of matrix--vector multiplication.\\
    Complexity: $\mathcal{O}(k \cdot |S|^2)$
    \;(or $\mathcal{O}(k \cdot |\text{transitions}|)$ with sparse matrices).
\end{frame}

\begin{frame}{PCTL model checking --- Unbounded until (the hard case)}

    \textbf{Goal:} check $P_{\sim p}[\phi_1 \, U\, \phi_2]$.

    \vspace{0.5em}

    \textbf{Step 1: Precomputation} (graph analysis, same as CTL):

    \begin{itemize}
        \item $S^{\text{no}} = \{s \mid \mathrm{Prob}(s,\, \phi_1 U \phi_2) = 0\}$
              \;---\; $\phi_2$ unreachable via $\phi_1$--states
        \item $S^{\text{yes}} = \{s \mid \mathrm{Prob}(s,\, \phi_1 U \phi_2) = 1\}$
              \;---\; reach $\phi_2$ almost surely
    \end{itemize}

    Both found by \textbf{backward graph search} (BFS/DFS).

    \vspace{0.8em}

    \textbf{Step 2: Solve} a linear equation system for the remaining states
    $S^{?} = S \setminus (S^{\text{no}} \cup S^{\text{yes}})$:

\end{frame}

\begin{frame}{}
    \begin{center}
        \[
            \boxed{
                x_s = \sum_{s' \in S} \mathbf{P}(s, s') \cdot x_{s'}
                \qquad \text{with } x_s = 0 \text{ for } s \in S^{\text{no}},\;\;
                x_s = 1 \text{ for } s \in S^{\text{yes}}
            }
        \]
    \end{center}

    \vspace{0.5em}

    \textbf{Step 3:} Check $x_s \sim p$ for each state $s$.

    \vspace{0.5em}
    {\footnotesize
        Solving methods: Gaussian elimination (exact)
        or iterative (Jacobi, Gauss--Seidel --- used in PRISM).
    }
\end{frame}

\begin{frame}{PCTL model checking --- CTL vs PCTL comparison}

    \begin{center}
        \renewcommand{\arraystretch}{1.3}
        \resizebox{0.95\textwidth}{!}{%
        \begin{tabular}{@{}l l l@{}}
            \toprule
            \textbf{Aspect}    & \textbf{CTL}            & \textbf{PCTL}                      \\
            \midrule
            Path quantifiers   & $\forall$, $\exists$    & $P_{\sim p}$                       \\
            Core computation   & graph fixpoint (BFS/DFS) & numerical (matrix ops + linear systems) \\
            Model              & Kripke structure        & DTMC (or MDP)                      \\
            Result per state   & Boolean                 & probability, then compare against $p$ \\
            \midrule
            \multicolumn{3}{l}{\textbf{Complexity:}} \\
            CTL                & \multicolumn{2}{l}{$\mathcal{O}(|\phi| \cdot (|S| + |R|))$
                \;---\; graph traversal}                                                      \\
            PCTL               & \multicolumn{2}{l}{$\mathcal{O}(|\phi| \cdot
                    (\text{poly}(|S|) + \text{equation solving}))$}                           \\
            \bottomrule
        \end{tabular}
        }
    \end{center}

    \vspace{1em}

    The shift can be summarized:\\
    CTL asks ``does this hold on all/some paths?''\\
    PCTL asks ``does the \textbf{probability} of this holding meet a threshold?''

    \vspace{0.5em}
    \hfill{\scriptsize\textcolor{gray}{[Baier \& Katoen, \emph{Principles of Model Checking}, 2008]}}
\end{frame}

\begin{frame}{Extension to MDPs}

    For \textbf{Markov Decision Processes} (nondeterminism + probability):

    \vspace{0.5em}

    \begin{itemize}
        \item Same PCTL syntax, but semantics must account for
              \textbf{all possible schedulers} (adversaries)

        \item Model checking computes \textbf{min/max probabilities}:
              \[
                  \Pr^{\min}_s(\psi) \quad\text{and}\quad \Pr^{\max}_s(\psi)
              \]

        \item For unbounded until: solve a \textbf{linear program} (LP)
              instead of a linear equation system

        \item Key result: \emph{memoryless deterministic} schedulers suffice
              for reachability properties
    \end{itemize}

    \vspace{1em}

    \hfill{\scriptsize\textcolor{gray}{[Parker, Lecture~14; Baier \& Katoen, Ch.~10.6]}}
\end{frame}

\begin{frame}{PRISM: The Foundational Model Checker}
\textbf{PRISM} model checker, primarily developed at the University of Oxford,
is the most widely used tool in the field that supports various types of models, including:
\begin{itemize}
    \item Discrete-Time Markov Chains (DTMCs)
    \item Markov Decision Processes (MDPs)
    \item Continuous-Time Markov Chains (CTMCs)
    \item Probabilistic Timed Automata (PTAs)
\end{itemize}
\end{frame}
\begin{frame}
    Its primary strength lies in its use of symbolic data structures,
    specifically Binary Decision Diagrams (BDDs) and Multi-Terminal Binary
    Decision Diagrams (MTBDDs).
    These structures allow PRISM to represent and analyze models
    with billions of states by exploiting the structural regularity
    of the system.
\end{frame}

\begin{frame}
    PRISM includes several analysis engines:
    \begin{itemize}
        \item Symbolic Engine: Uses MTBDDs for both representation and numerical solution.
        \item  Explicit Engine: Uses sparse matrices and is often faster for models with less regularity.
        \item Hybrid Engine: Combines the memory efficiency of MTBDDs with the speed of sparse matrices.
        \item Simulator: Supports Statistical Model Checking (SMC), which uses Monte Carlo simulation
        to estimate probabilities when the state space is too large for exhaustive analysis.
    \end{itemize}

    PRISM-games extends its capabilities to stochastic multi-player games (systems with multiple agents)
\end{frame}

\begin{frame}{Diverse Application Domains}
    The versatility of probabilistic model checking has led to its adoption across
     a wide range of application domains, including:
    \begin{itemize}
        \item Randomised distributed algorithms
        \item Communication protocols and networks
        \item Computer security
        \item Autonomous systems and robotics
        \item Safety-critical systems
        \item Biology and human behavior
        \item AI and self-adaptive systems
        \item Many more...
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Example 1: Message Delivery Protocol (DTMC)}
    \textbf{Scenario:} A sender transmits a message that is lost with probability $0.3$.
    The sender retries up to 3 times.

    \vspace{0.5em}

    \begin{semiverbatim}
dtmc
module protokol
  s       : [0..2] init 0; 
  retries : [0..3] init 0;

  [] s=0 & retries<3 ->
       0.7 : (s'=1) + 0.3 : (s'=0) & (retries'=retries+1);
  [] s=0 & retries=3 -> 1 : (s'=2);
  [] s=1 -> 1 : (s'=1);
  [] s=2 -> 1 : (s'=2);
endmodule

label "delivered" = s=1;
label "failure"   = s=2;
    \end{semiverbatim}
\end{frame}

\begin{frame}{Example 1: Properties and Results}
    \textbf{PCTL properties we verify:}

    \vspace{0.5em}

    \small
    \begin{tabular}{p{5cm}p{3.8cm}l}
        \toprule
        \textbf{Property} & \textbf{Question} & \textbf{Result} \\
        \midrule
        \texttt{P=? [ F "delivered" ]}    & Probability of delivery?        & $0.973$ \\
        \texttt{P=? [ F "failure" ]}      & Probability of total failure?   & $0.027$ \\
        \texttt{P=? [ F<=1 "delivered" ]} & Delivered on first try?         & $0.7$   \\
        \texttt{P>=0.95 [ F "delivered"]} & Is reliability above 95\%?      & \texttt{true} \\
        \bottomrule
    \end{tabular}

    \vspace{1em}

    Note: \texttt{P=? [ F "failure" ]} $= 0.3^3 = 0.027$ — PRISM computes the
    \textbf{exact} mathematical value, not a simulation estimate.
\end{frame}

\begin{frame}[fragile]{Example 2: Football Club (MDP)}
    \textbf{Scenario:} Each season, the board chooses to \emph{organise a tournament}
    or \emph{do nothing}. The outcome is still probabilistic.

    \vspace{0.5em}

    \begin{semiverbatim}
mdp
module club
  s : [0..1] init 0; \textit{// 0=struggling, 1=thriving}

  [tournament] s=0 -> 0.6:(s'=1) + 0.4:(s'=0);
  [tournament] s=1 -> 0.9:(s'=1) + 0.1:(s'=0);
  [nothing]    s=0 -> 0.2:(s'=1) + 0.8:(s'=0);
  [nothing]    s=1 -> 0.6:(s'=1) + 0.4:(s'=0);
endmodule

label "struggling" = s=0;
label "thriving"   = s=1;
    \end{semiverbatim}
\end{frame}
