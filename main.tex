\begin{frame}{Motivation}
    We want to model a probabilistic system and check its properties.

    Examples include:
    \begin{itemize}
        \item Randomised distributed algorithms
        \item Communication, network and multimedia protocols
        \item Security protocols
        \item Biological systems
        \item Planning and synthesis
        \item Performance and reliability
        \item Game theory
        \item Power management
        \item ...
    \end{itemize}
\end{frame}

\begin{frame}{How do we model a probabilistic system?}
    \begin{enumerate}
        \item describe states and transitions
        \item make number of state finite by abstraction
        \item we assign probabilities to transitions
    \end{enumerate}
\end{frame}
